// Skills matrix: select a skill row to highlight the projects that used it.
const table = document.querySelector("#skills-matrix");
const heads = [...table.querySelectorAll("thead th")].slice(1);
for (const row of table.querySelectorAll("tbody tr")) {
  const btn = row.querySelector("button");
  btn.addEventListener("click", () => {
    const on = !row.classList.contains("sel");
    table.querySelectorAll("tbody tr").forEach((r) => { r.classList.remove("sel"); r.querySelector("button").setAttribute("aria-pressed", "false"); });
    heads.forEach((h) => h.classList.remove("on"));
    if (on) {
      row.classList.add("sel");
      btn.setAttribute("aria-pressed", "true");
      [...row.dataset.flags].forEach((f, j) => { if (f === "1") heads[j].classList.add("on"); });
    }
  });
}
